cmake_minimum_required(VERSION 3.22)
project(predictor_native_checkpoint LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_GLES ON CACHE BOOL "" FORCE)
set(ENABLE_PLAYLIST OFF CACHE BOOL "" FORCE)
set(ENABLE_SYSTEM_PROJECTM_EVAL OFF CACHE BOOL "" FORCE)
set(ENABLE_SDL_UI OFF CACHE BOOL "" FORCE)
set(ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_FLEX ON)
set(CMAKE_DISABLE_FIND_PACKAGE_BISON ON)
add_subdirectory("${PROJECTM_SOURCE}" projectm)
add_executable(native-ruler native-ruler.cpp)
target_compile_features(native-ruler PRIVATE cxx_std_17)
target_compile_definitions(native-ruler PRIVATE USE_GLES)
target_include_directories(native-ruler PRIVATE "${PROJECTM_SOURCE}/src/libprojectM" "${PROJECTM_SOURCE}/vendor/glm" "${PROJECTM_SOURCE}/vendor/glad/include" "${JSON_INCLUDE}")
target_link_libraries(native-ruler PRIVATE libprojectM::projectM)
